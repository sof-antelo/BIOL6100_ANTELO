Part 1: Create a funtion that takes the parameters required to run a logistic growth model and returns a dataframe with columns for population size and time.

####################
#Start of Function

log_growth_func <- function(N0 = 1, r = .1, K = 100, tf = 100, ts = 1) {
#####################
# FUNCTION: log_growth_func
# PURPOSE:Logistic growth model for population growth
# INPUTS: Initial population size (N0), growth rate(r), carrying capacity (K), end time (tf), timestep (ts)
# OUT: dataframe with population size and time

func_time <- seq(from = 0, to = tf, by = ts)
N_t <- (K/(1+((K-N0)/N0)*exp((-r)*func_time)))

my_df <- data.frame(x=func_time, y=N_t)

return(my_df)
}
#####################
