Mapping with Plotly

PART 1: Choropleth map of GDP in Europe

from bokeh.palettes import viridis
import json 
from networkx.algorithms.bipartite import projection
import pandas as pd
import plotly.express as px
import plotly.graph_objects as go

#loading data, filtering for Europe and a specific year, I am using 2002
gap = px.data.gapminder().query("continent=='Europe' and year == 2002")
gap.head()

#making choropleth map
fig = px.choropleth(
     gap,
    locations = "iso_alpha",
    color = "gdpPercap",
    hover_name = "country",
    hover_data = {
        "pop": ":,",
        "gdpPercap": ":,.2f",
        "iso_alpha": False
    },
    color_continuous_scale="viridis",
    title = "GDP per cap. by country (2002)",
    labels = {"country":"Country", "pop":"Population","gdpPercap":"GDP per Capita"}
)
#scope
fig.update_geos(
    scope = "europe",
    showland = True
)

fig.show()

#I used the additional argument 'scope'. 'Scope' essentially crops the map, so I cropped it to focus on Europe.

gap = px.data.gapminder().query("year == 2002")
fig = px.scatter_geo(
    gap,
    locations="iso_alpha",
    size="pop",
    color="continent",
    hover_name="country",
    hover_data={
        "pop": ":,",
        "lifeExp": ":.1f",

    }
)

fig.show()
fig2 = px.scatter_geo(
    gap,

    hover_name="country",
    hover_data={
        "pop": ":,",

    },

)
fig2.show()
#using the argument projection = orthographic changes the layout of the map so that it is a globe rather than a flat map, more accurate
