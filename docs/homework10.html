PART 3: Create a numpy array of length 100 with values bound between 0 and 1 and save as arr1.

from numpy import random
# Numpy array length = 100, values 0 - 1
arr1 = np.random.rand(100)

#extract the last 10 items with negative indexing
neg_indx = arr1[-10: -1]
print("Last 10 elements:", neg_indx)

# convert numeric floats to strings and back to floats
arr_str = arr1.astype(str)         # strings
arr_float = arr_str.astype(float)  #  back to floats
print("First 5 elements after conversion:", arr_float[:5])

# reshape array into a 10x10 2D array
arr2 = arr1.reshape(10, 10)

# create a 10x10 binary array with probability 0.2 for 1s and 0.8 for 0s
arr3 = np.random.choice([0, 1], size=(10, 10), p=[0.8, 0.2])
